\section{Case 1: Annuity Valuation (Mr.\ Thompson)}

\textbf{Assignment.} Value three annuity structures at $r=6\%$ and recommend the highest present value.

\subsection{Methodology}
\begin{align}
  PV_{\text{ordinary}} &= A \cdot \frac{1-(1+r)^{-N}}{r} \\
  PV_{\text{delayed}} &= \frac{PV_{@Y10}}{(1+r)^{10}}, \quad PV_{@Y10} = A \cdot \frac{1-(1+r)^{-15}}{r} \\
  PV_{\text{growing perpetuity}} &= \frac{C}{r-g}, \quad r > g
\end{align}

\section{Case 2: OceanBlue Fund (Brooks)}

\textbf{Assignment.} Compute money-weighted and time-weighted returns; state which is higher.

\subsection{Methodology}
Money-weighted return (MWRR) solves $\sum_{t=0}^{T} CF_t/(1+\mathrm{MWRR})^t = 0$ on investor cash flows (contributions at the beginning of each year per workbook). Time-weighted return (TWRR) is the geometric mean of annual net returns: $\sqrt[T]{\prod(1+R_t)}-1$.

\subsection{Results}
\begin{itemize}[noitemsep]
  \item \textbf{MWRR} = \CaseTwoMWRR{}
  \item \textbf{TWRR} = \CaseTwoTWRR{}
  \item \textbf{Higher return:} Time-weighted
  \item \textbf{Sanity check:} $CF_0 = -\$\CaseTwoCFYzero$M, $CF_1 = -\$\CaseTwoCFYone$M (required $-$\$75$M and $-$\$15$M)
\end{itemize}

\subsection{Interpretation}
TWRR exceeds MWRR: the Brooks family's contribution and withdrawal \emph{timing} reduced their realized dollar return relative to the fund's pure performance series.
